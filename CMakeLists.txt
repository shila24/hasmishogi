cmake_minimum_required(VERSION 3.16)
project(hasami_lab VERSION 1.0.0 LANGUAGES CXX)
option(HASAMI_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hasami_core STATIC cpp/src/game.cpp cpp/src/search.cpp)
set_target_properties(hasami_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(hasami_core PUBLIC cpp/include)
add_library(hasami_native SHARED cpp/src/c_api.cpp)
target_link_libraries(hasami_native PRIVATE hasami_core)
target_compile_definitions(hasami_native PRIVATE HASAMI_BUILD)
add_executable(hasami_cli cpp/src/main.cpp)
target_link_libraries(hasami_cli PRIVATE hasami_core)
add_executable(hasami_native_tests cpp/tests/test_core.cpp)
target_link_libraries(hasami_native_tests PRIVATE hasami_core hasami_native)
# Native riichi engine (riichi/native.py). Sources are globbed so worker tasks add
# files without editing this orchestrator-owned file: cpp/riichi/src/c_api*.cpp form the shared C ABI
# library, every other cpp/riichi/src/*.cpp the static core, each cpp/riichi/tests/*.cpp one ctest
# executable, and each cpp/riichi/tools/*.cpp one tool executable.
file(GLOB RIICHI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/riichi/src/*.cpp)
set(RIICHI_ABI_SOURCES ${RIICHI_SOURCES})
list(FILTER RIICHI_ABI_SOURCES INCLUDE REGEX "/c_api[^/]*\\.cpp$")
list(FILTER RIICHI_SOURCES EXCLUDE REGEX "/c_api[^/]*\\.cpp$")
set(RIICHI_TARGETS)
if(RIICHI_SOURCES)
    add_library(riichi_core STATIC ${RIICHI_SOURCES})
    set_target_properties(riichi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(riichi_core PUBLIC cpp/riichi/include)
    list(APPEND RIICHI_TARGETS riichi_core)
    if(RIICHI_ABI_SOURCES)
        add_library(riichi_native SHARED ${RIICHI_ABI_SOURCES})
        target_link_libraries(riichi_native PRIVATE riichi_core)
        target_compile_definitions(riichi_native PRIVATE RIICHI_BUILD)
        list(APPEND RIICHI_TARGETS riichi_native)
    endif()
    file(GLOB RIICHI_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/riichi/tests/*.cpp)
    foreach(source ${RIICHI_TEST_SOURCES})
        get_filename_component(name ${source} NAME_WE)
        add_executable(riichi_${name} ${source})
        target_link_libraries(riichi_${name} PRIVATE riichi_core)
        list(APPEND RIICHI_TARGETS riichi_${name})
        add_test(NAME riichi_${name} COMMAND riichi_${name})
    endforeach()
    file(GLOB RIICHI_TOOL_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/riichi/tools/*.cpp)
    foreach(source ${RIICHI_TOOL_SOURCES})
        get_filename_component(name ${source} NAME_WE)
        add_executable(riichi_${name} ${source})
        target_link_libraries(riichi_${name} PRIVATE riichi_core)
        list(APPEND RIICHI_TARGETS riichi_${name})
    endforeach()
endif()
# Native shogi engine (shogi/native.py), globbed exactly like riichi: cpp/shogi/src/c_api*.cpp form
# the shared C ABI library shogi_native, every other cpp/shogi/src/*.cpp the static core, each cpp/shogi/tests/*.cpp one
# ctest executable and each cpp/shogi/tools/*.cpp one tool executable.
file(GLOB SHOGI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/shogi/src/*.cpp)
set(SHOGI_ABI_SOURCES ${SHOGI_SOURCES})
list(FILTER SHOGI_ABI_SOURCES INCLUDE REGEX "/c_api[^/]*\\.cpp$")
list(FILTER SHOGI_SOURCES EXCLUDE REGEX "/c_api[^/]*\\.cpp$")
set(SHOGI_TARGETS)
if(SHOGI_SOURCES)
    add_library(shogi_core STATIC ${SHOGI_SOURCES})
    set_target_properties(shogi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(shogi_core PUBLIC cpp/shogi/include)
    list(APPEND SHOGI_TARGETS shogi_core)
    if(SHOGI_ABI_SOURCES)
        add_library(shogi_native SHARED ${SHOGI_ABI_SOURCES})
        target_link_libraries(shogi_native PRIVATE shogi_core)
        target_compile_definitions(shogi_native PRIVATE SHOGI_BUILD)
        list(APPEND SHOGI_TARGETS shogi_native)
    endif()
    file(GLOB SHOGI_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/shogi/tests/*.cpp)
    foreach(source ${SHOGI_TEST_SOURCES})
        get_filename_component(name ${source} NAME_WE)
        add_executable(shogi_${name} ${source})
        target_link_libraries(shogi_${name} PRIVATE shogi_core)
        list(APPEND SHOGI_TARGETS shogi_${name})
        add_test(NAME shogi_${name} COMMAND shogi_${name})
    endforeach()
    file(GLOB SHOGI_TOOL_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cpp/shogi/tools/*.cpp)
    foreach(source ${SHOGI_TOOL_SOURCES})
        get_filename_component(name ${source} NAME_WE)
        add_executable(shogi_${name} ${source})
        target_link_libraries(shogi_${name} PRIVATE shogi_core)
        list(APPEND SHOGI_TARGETS shogi_${name})
    endforeach()
endif()
foreach(target hasami_core hasami_native hasami_cli hasami_native_tests ${RIICHI_TARGETS} ${SHOGI_TARGETS})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(HASAMI_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
enable_testing()
add_test(NAME native_core COMMAND hasami_native_tests)

# Independent batched PUCT, linked directly to the game cores, never their C ABIs.
find_package(Threads REQUIRED)
add_library(mcts_native SHARED cpp/mcts/src/c_api.cpp)
target_include_directories(mcts_native PUBLIC cpp/mcts/include)
target_link_libraries(mcts_native PRIVATE hasami_core shogi_core Threads::Threads)
target_compile_definitions(mcts_native PRIVATE MCTS_BUILD)
add_executable(mcts_tests cpp/mcts/tests/test_search.cpp)
target_include_directories(mcts_tests PRIVATE cpp/mcts/include)
target_link_libraries(mcts_tests PRIVATE mcts_native hasami_core shogi_core Threads::Threads)
foreach(target mcts_native mcts_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
        if(HASAMI_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME native_mcts COMMAND mcts_tests)

# Family N owns its search position and links the reference core only for input
# validation and parity tests. No external chess/shogi engine is linked.
add_library(shogi_search_core STATIC
    cpp/shogi_search/src/position.cpp cpp/shogi_search/src/search.cpp
    cpp/shogi_search/src/nnue.cpp cpp/shogi_search/src/binary.cpp cpp/shogi_search/src/data.cpp)
set_target_properties(shogi_search_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(shogi_search_core PUBLIC cpp/shogi_search/include)
target_link_libraries(shogi_search_core PUBLIC shogi_core)
add_library(shogi_search_native SHARED cpp/shogi_search/src/c_api.cpp)
target_link_libraries(shogi_search_native PRIVATE shogi_search_core)
target_compile_definitions(shogi_search_native PRIVATE SHOGI_SEARCH_BUILD)

    add_executable(shogi_search_${name} ${source})

    target_compile_definitions(shogi_search_${name} PRIVATE SS_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME shogi_search_${name} COMMAND shogi_search_${name})
    list(APPEND SHOGI_SEARCH_TARGETS shogi_search_${name})
endforeach()
file(GLOB SHOGI_SEARCH_TOOLS CONFIGURE_DEPENDS cpp/shogi_search/tools/*.cpp)

    target_link_libraries(shogi_search_${name} PRIVATE shogi_search_core Threads::Threads)

endforeach()

            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
